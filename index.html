    async function connectToRoom() {

      if (!selectedRole) {

        setStatus(

          "Please choose Teacher or Student first.",

          "error"

        );

        return;

      }


      const room = sessionInput.value.trim();


      if (room.length < 4) {

        setStatus(

          "Please enter a Session Name with at least 4 characters.",

          "error"

        );

        return;

      }


      /*

        Close a previous connection if the user changes rooms.

      */

      if (mqttClient) {

        try {

          mqttClient.end(true);

        } catch (error) {

          console.warn(error);

        }

      }


      currentRoom = room;


      setStatus(

        "Preparing encrypted classroom connection…",

        "normal"

      );


      joinBtn.disabled = true;

      joinBtn.textContent = "Connecting…";


      try {

        /*

          Derive both:

          1. A non-obvious MQTT topic.

          2. An encryption key.


          The real Session Name is therefore never used as

          the broker topic.

        */

        topicName = await createHashedTopic(currentRoom);

        encryptionKey = await deriveEncryptionKey(currentRoom);


        openMqttConnection();


      } catch (error) {

        console.error(error);


        setStatus(

          "Unable to prepare this room. Try another Session Name.",

          "error"

        );


        joinBtn.disabled = false;

        joinBtn.textContent = "Connect";

      }

    }



    function openMqttConnection() {

      const clientId =

        "class_translate_" +

        crypto.randomUUID().replaceAll("-", "");


      mqttClient = mqtt.connect(

        BROKER_URL,

        {

          clientId,

          clean: true,

      mqttClient.on(

        "message",

        async (topic, payload) => {

          if (topic !== topicName) {

            return;

          }


          try {

            const encryptedMessage =

              payload.toString();


            const decoded =

              await decryptMessage(encryptedMessage);


            if (!decoded) {

              return;

            }


            const data = JSON.parse(decoded);


            /*

              Only the Student UI consumes classroom translation

              messages.

            */

            if (

              selectedRole === "student" &&

              data.type === "translation" &&

              data.spanish

            ) {

              updateStudentDisplay(data);

            }


          } catch (error) {

            /*

              Messages on a public broker can come from elsewhere.

              Any payload that cannot be decrypted with this

              classroom's key is simply ignored.

            */

            console.warn(

              "Ignored unrelated/invalid MQTT message."

            );

          }

        }

      );

    }

    /* ==========================================================

       HASH ROOM NAME INTO MQTT TOPIC

       ========================================================== */


    async function createHashedTopic(roomName) {

      const encoder = new TextEncoder();


      /*

        App-specific prefix stops a plain SHA hash lookup of

        "Room101" from directly matching the topic.

      */

      const input =

        encoder.encode(

          "live-classroom-spanish-v1::" + roomName

        );


      const hash =

        await crypto.subtle.digest(

          "SHA-256",

          input

        );


      const bytes =

        Array.from(new Uint8Array(hash));


      const hex =

        bytes

          .map(byte =>

            byte.toString(16).padStart(2, "0")

          )

          .join("");


      /*

        Namespace unique to this app.

      */

      return "live-classroom-translator/v1/" + hex;

    }



    /* ==========================================================

       CLIENT-SIDE ENCRYPTION

       ========================================================== */


    async function deriveEncryptionKey(roomName) {

      const encoder = new TextEncoder();


      const baseKey =

        await crypto.subtle.importKey(

          "raw",

          encoder.encode(roomName),

          "PBKDF2",

          false,

          ["deriveKey"]

        );


      return crypto.subtle.deriveKey(

        {

          name: "PBKDF2",


          /*

            A static application salt is acceptable here because

            the Session Name itself is the shared secret.

          */

          salt: encoder.encode(

            "live-classroom-translator-encryption-v1"

          ),


          iterations: 120000,

          hash: "SHA-256"

        },

        baseKey,

        {

          name: "AES-GCM",

          length: 256

        },

        false,

        [

          "encrypt",

          "decrypt"

        ]

      );

    }



    async function encryptMessage(plainText) {

      const encoder = new TextEncoder();


      const iv =

        crypto.getRandomValues(

          new Uint8Array(12)

        );


      const encrypted =

        await crypto.subtle.encrypt(

          {

            name: "AES-GCM",

            iv

          },

          encryptionKey,

          encoder.encode(plainText)

        );


      return JSON.stringify({

        iv: arrayBufferToBase64(iv),

        data: arrayBufferToBase64(encrypted)

      });

    }



    async function decryptMessage(payload) {

      const parsed = JSON.parse(payload);


      const iv =

        base64ToUint8Array(parsed.iv);


      const ciphertext =

        base64ToUint8Array(parsed.data);


      const decrypted =

        await crypto.subtle.decrypt(

          {

            name: "AES-GCM",

            iv

          },

          encryptionKey,

          ciphertext

        );


      return new TextDecoder().decode(decrypted);

    }



    function arrayBufferToBase64(buffer) {

      const bytes =

        new Uint8Array(buffer);


      let binary = "";


      for (let i = 0; i < bytes.byteLength; i++) {

        binary += String.fromCharCode(bytes[i]);

      }


      return btoa(binary);

    }



    function base64ToUint8Array(base64) {

      const binary =

        atob(base64);


      const bytes =

        new Uint8Array(binary.length);


      for (let i = 0; i < binary.length; i++) {

        bytes[i] =

          binary.charCodeAt(i);

      }


      return bytes;

    }



    /* ==========================================================

       TEACHER: SPEECH RECOGNITION

       ========================================================== */


    function setupSpeechRecognition() {

      if (recognition) {

        return;

      }


      const SpeechRecognition =

        window.SpeechRecognition ||

        window.webkitSpeechRecognition;


      if (!SpeechRecognition) {

        micBtn.disabled = true;


        micBtn.style.opacity = ".4";


        micLabel.textContent =

          "Speech recognition is not available in this browser. Use the text box below.";


        return;

      }


      recognition =

        new SpeechRecognition();


      recognition.lang = "en-US";

      recognition.continuous = true;

      recognition.interimResults = true;



      recognition.onstart = () => {

        isListening = true;


        micBtn.classList.add("listening");


        micLabel.textContent =

          "Listening… tap again to stop";

      };



      recognition.onresult = event => {

        let interimTranscript = "";

        let finalTranscript = "";


        for (

          let i = event.resultIndex;

          i < event.results.length;

          i++

        ) {

          const transcript =

            event.results[i][0].transcript.trim();


          if (event.results[i].isFinal) {

            finalTranscript +=

              (finalTranscript ? " " : "") +

              transcript;

          } else {

            interimTranscript +=

              (interimTranscript ? " " : "") +

              transcript;

          }

        }


        /*

          Show interim speech instantly on the teacher screen,

          but do NOT repeatedly call the free translation API

          for every partial word.

        */

        if (interimTranscript) {

          speechPreview.textContent =

            interimTranscript + " …";

        }


        if (finalTranscript) {

          speechPreview.textContent =

            finalTranscript;


          /*

            Avoid accidental duplicate final events.

          */

          const now = Date.now();


          const duplicate =

            finalTranscript === lastSpeechText &&

            now - lastSpeechTime < 2500;


          if (!duplicate) {

            lastSpeechText = finalTranscript;

            lastSpeechTime = now;


            translateAndBroadcast(

              finalTranscript,

              "speech"

            );

          }

        }

      };



      recognition.onerror = event => {

        console.warn(

          "Speech recognition error:",

          event.error

        );


        if (

          event.error === "not-allowed" ||

          event.error === "service-not-allowed"

        ) {

          isListening = false;


          micBtn.classList.remove("listening");


          micLabel.textContent =

            "Microphone permission was blocked. You can still type below.";

        }

      };



      recognition.onend = () => {

        /*

          Chrome can automatically stop continuous recognition

          after a while. If the teacher still considers the mic

          active, attempt to restart it.

        */

        if (isListening) {

          try {

            recognition.start();

          } catch (error) {

            console.warn(error);

          }

        } else {

          micBtn.classList.remove("listening");


          micLabel.textContent =

            "Tap to start speaking";

        }

      };

    }



    micBtn.addEventListener("click", () => {

      if (!recognition) {

        setupSpeechRecognition();

      }


      if (!recognition) {

        return;

      }


      if (isListening) {

        isListening = false;


        try {

          recognition.stop();

        } catch (error) {

          console.warn(error);

        }


        micBtn.classList.remove("listening");


        micLabel.textContent =

          "Tap to start speaking";


      } else {

        try {

          recognition.start();

        } catch (error) {

          console.warn(error);

        }

      }

    });

    /* ==========================================================

       TRANSLATE ENGLISH → SPANISH

       ========================================================== */


    async function translateToSpanish(englishText) {

      /*

        MyMemory:

          q        = text to translate

          langpair = source|target

      */


      const url =

        `${TRANSLATE_URL}?q=${encodeURIComponent(englishText)}&langpair=en|es`;


      const response =

        await fetch(url);


      if (!response.ok) {

        throw new Error(

          "Translation service returned " +

          response.status

        );

      }


      const data =

        await response.json();


      if (

        !data ||

        !data.responseData ||

        !data.responseData.translatedText

      ) {

        throw new Error(

          "Translation response was incomplete."

        );

      }


      return decodeHtmlEntities(

        data.responseData.translatedText

      );

    }



    /*

      Translation APIs can occasionally encode apostrophes,

      quotation marks, etc. as HTML entities.

    */

    function decodeHtmlEntities(text) {

      const textarea =

        document.createElement("textarea");


      textarea.innerHTML = text;


      return textarea.value;

    }



    /* ==========================================================

       TRANSLATE + REALTIME BROADCAST

       ========================================================== */


    async function translateAndBroadcast(

      englishText,

      source

    ) {

      if (

        !mqttClient ||

        !mqttClient.connected ||

        !encryptionKey

      ) {

        setStatus(

          "The classroom is not connected yet.",

          "error"

        );


        return;

      }


      const cleanText =

        englishText.trim();


      if (!cleanText) {

        return;

      }


      teacherSpanishPreview.textContent =

        "Translating…";


      sendBtn.classList.add("loading");


      try {

        const spanish =

          await translateToSpanish(cleanText);


        teacherSpanishPreview.textContent =

          spanish;


        const message = {

          type: "translation",

          source,

          english: cleanText,

          spanish,

          timestamp: Date.now()

        };


        /*

          Encrypt BEFORE publishing.

        */

        const encryptedPayload =

          await encryptMessage(

            JSON.stringify(message)

          );


        mqttClient.publish(

          topicName,

          encryptedPayload,

          {

            /*

              QoS 1 = at least once delivery.

              The student code safely tolerates an occasional

              duplicate.

            */

            qos: 1,


            /*

              Do not retain classroom speech on the broker.

            */

            retain: false

          },

          error => {

            if (error) {

              console.error(error);


              setStatus(

                "Translation completed but could not be sent.",

                "error"

              );

            }

          }

        );


      } catch (error) {

        console.error(error);


        teacherSpanishPreview.textContent =

          "Translation unavailable. Please try again.";


      } finally {

        sendBtn.classList.remove("loading");

      }

    }

        "Received " +

        date.toLocaleTimeString(

          [],

          {

            hour: "numeric",

            minute: "2-digit",

            second: "2-digit"

          }

        );

    }



    /* ==========================================================

       STUDENT TEXT-TO-SPEECH

       ========================================================== */


    speakerBtn.addEventListener("click", () => {

      if (!latestSpanish) {

        return;

      }


      /*

        Stop previous speech before starting another.

      */

      window.speechSynthesis.cancel();


      const utterance =

        new SpeechSynthesisUtterance(

          latestSpanish

        );


      utterance.lang = "es-ES";

      utterance.rate = 0.9;

      utterance.pitch = 1;


      /*

        Try to use an installed Spanish voice when available.

      */

      const voices =

        window.speechSynthesis.getVoices();


      const spanishVoice =

        voices.find(voice =>

          voice.lang

            .toLowerCase()

            .startsWith("es")

        );


      if (spanishVoice) {

        utterance.voice =

          spanishVoice;

      }


      window.speechSynthesis.speak(

        utterance

      );

    });
